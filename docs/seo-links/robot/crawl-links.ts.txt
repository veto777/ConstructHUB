// Owner's rule 2026-10-09: every figure on every SEO screen is a link that lands on its data with the filter shown.
// This visits each SEO start page on the rig, collects every in-app SEO link inside <main>, follows each one once and
// reports: a page that errors, says "not found", scrolls sideways, lacks the page's own content, or carries a filter
// parameter without showing the active-filter chip. Usage: npx tsx scripts/crawl-links.tmp.ts [maxPerPage]
import { chromium } from "playwright";

const BASE = "http://127.0.0.1:8189";
const START = [
  "/seo", "/seo/rank-tracker?site=3", "/seo/rank-tracker?site=1", "/seo/explorer?domain=jameshardie.com", "/seo/audit?site=1",
  "/seo/keywords?keyword=hardie%20plank%20siding", "/seo/content?q=fiber%20cement%20siding%20cost", "/seo/backlinks?site=1",
  "/seo/reports?site=1", "/seo/local-grid?site=1", "/seo/ai?site=1", "/seo/alerts", "/seo/plan?site=1", "/seo/mentions?site=1",
  "/seo/usage", "/seo/batch", "/seo/competitors?site=1",
];
// Parameters that only choose the subject, not a narrowing (no chip expected for these alone).
const SUBJECT = new Set(["site", "domain", "keyword", "q", "view", "tab", "panel", "section", "scan", "locationCode", "table", "at", "credits", "group", "order", "sort"]);
const MAX = Number(process.argv[2] ?? 40);

type Result = { from: string; href: string; status: string };
(async () => {
  const b = await chromium.launch({ headless: true, args: ["--disable-dev-shm-usage"] });
  const ctx = await b.newContext({ viewport: { width: 1280, height: 2000 } });
  await ctx.addInitScript("localStorage.setItem('hub.welcomeSeen','1')");
  let page = await ctx.newPage();
  const errors: string[] = [];
  const wire = (pg: typeof page) => { pg.on("pageerror", (e) => errors.push(e.message.slice(0, 120))); pg.on("crash", () => errors.push("PAGE CRASHED")); };
  wire(page);
  // A page that keeps navigating after it loaded (an address rewrite on arrival) destroys the evaluate context: that is a finding, not a crash.
  const ev = async <T,>(expr: string, fallback: T): Promise<T> => { try { return (await page.evaluate(expr)) as T; } catch { return fallback; } };
  const fresh = async () => { try { await page.close(); } catch { /* gone */ } page = await ctx.newPage(); wire(page); };
  const seen = new Set<string>(); const results: Result[] = []; let total = 0, linksFound = 0;
  for (const start of START) {
    let hrefs: string[] = [];
    try {
      await page.goto(BASE + start, { waitUntil: "domcontentloaded", timeout: 30000 });
      await page.waitForTimeout(4000);
      hrefs = await ev<string[]>("Array.from(document.querySelectorAll('main a[href^=\"/seo\"]')).map(a => a.getAttribute('href'))", []);
      console.log(`page ${start}: ${new Set(hrefs).size} links`);
    } catch (e) { results.push({ from: start, href: start, status: "START CRASH " + String(e).slice(0, 100) }); await fresh(); continue; }
    const uniq = [...new Set(hrefs)].filter((h) => h && !seen.has(h)).slice(0, MAX);
    linksFound += [...new Set(hrefs)].length;
    for (const href of uniq) {
      seen.add(href); total++;
      errors.length = 0;
      const problems: string[] = [];
      let r: Awaited<ReturnType<typeof page.goto>> = null;
      try {
        // In-app navigation (a click on the real link) is what a visitor does and is fast on the dev rig; a full
        // reload is the fallback when the link is no longer on the page after coming back.
        const a = page.locator(`main a[href="${href.replace(/"/g, '\\"')}"]`).first();
        if (await a.count()) { await a.scrollIntoViewIfNeeded().catch(() => null); await a.click({ timeout: 5000 }); await page.waitForURL((u) => u.toString().includes(href.split("?")[0]), { timeout: 8000 }).catch(() => null); await page.waitForTimeout(1200); r = null; }
        else { r = await page.goto(BASE + href, { waitUntil: "domcontentloaded", timeout: 20000 }); await page.waitForTimeout(1800); }
        if (page.url().replace(BASE, "") !== href && !page.url().includes(href.split("?")[0])) problems.push(`landed on ${page.url().replace(BASE, "")}`);
      } catch (e) {
        const msg = String(e).slice(0, 100);
        if (/crashed|closed|Target/i.test(msg) || errors.includes("PAGE CRASHED")) { results.push({ from: start, href, status: "CRASH " + msg }); await fresh(); continue; }
        problems.push("load: " + msg);
      }
      if (r && r.status() >= 400) problems.push(`http ${r.status()}`);
      const urlAfter = page.url(); await page.waitForTimeout(600);
      if (page.url() !== urlAfter) problems.push(`keeps navigating: ${page.url().replace(BASE, "")}`);
      const text = await ev<string>("document.body.innerText.slice(0, 20000)", "");
      if (!text) problems.push("no text (context lost)");
      if (/page not found|404/i.test(text.slice(0, 400))) problems.push("not found");
      if (errors.length) problems.push(`js: ${errors[0]}`);
      const overflow = await ev<number>("document.documentElement.scrollWidth - document.documentElement.clientWidth", 0);
      if (overflow > 0) problems.push(`overflow ${overflow}`);
      const u = new URL(BASE + href); const narrowing = [...u.searchParams.keys()].filter((k) => !SUBJECT.has(k));
      if (narrowing.length) {
        // The chip appears once the page's data has loaded: wait for it (up to 8 s) before calling it missing.
        let chip = await page.locator("[data-testid=active-filter]").count();
        for (let i = 0; i < 16 && !chip; i++) { await page.waitForTimeout(500); chip = await page.locator("[data-testid=active-filter]").count(); }
        if (!chip) problems.push(`no active-filter chip for ${narrowing.join(",")}`);
      }
      const bought = /charged|purchased just now/i.test(text) ? "" : "";
      if (bought) problems.push(bought);
      results.push({ from: start, href, status: problems.length ? problems.join("; ") : "ok" });
      // Back to the start page the fast way; a reload if the app did not come back.
      try { await page.goBack({ waitUntil: "domcontentloaded", timeout: 10000 }); await page.waitForTimeout(700); } catch { /* fall through */ }
      if (!page.url().includes(start.split("?")[0])) { try { await page.goto(BASE + start, { waitUntil: "domcontentloaded", timeout: 30000 }); await page.waitForTimeout(2500); } catch { /* next */ } }
    }
  }
  const bad = results.filter((r) => r.status !== "ok");
  console.log(`links found ${linksFound}, followed ${total}, problems ${bad.length}`);
  for (const r of bad) console.log(`BAD  ${r.status}  ${r.href}  (from ${r.from})`);
  await b.close();
})().catch((e) => { console.error("FAILED", String(e).slice(0, 400)); process.exit(1); });
